# Run on LIARA as root (crontab -e).  Liara PULLS the newest daily backup from UniWeb (port 22 is open Liara->UniWeb only).
# Weekly Mon 02:00 UTC; needs 2x backup size + 5 GB free on Liara first; keeps the 2 newest biap-global-YYYY-MM-DD copies; touches nothing else in uniweb-backups.
0 2 * * 1 flock -n /run/biap-pull.lock sh -c 'S=ubuntu@5.249.252.88; B=/home/ubuntu/backups/daily; D=/root/uniweb-backups; O="-i /root/.ssh/id_ed25519 -o BatchMode=yes -o IdentitiesOnly=yes -o ConnectTimeout=20"; mkdir -p $D; L=$(ssh $O $S "ls -1 $B | grep -E \"^[0-9]{4}-[0-9]{2}-[0-9]{2}\$\" | sort | tail -1"); [ -n "$L" ] || { echo "$(date -u +%FT%TZ) FAIL no backup/ssh"; exit 1; }; K=$(ssh $O $S "du -sk $B/$L | cut -f1"); F=$(df -Pk $D | awk "NR==2{print \$4}"); [ "$F" -ge $((K*2+5242880)) ] || { echo "$(date -u +%FT%TZ) FAIL low space free_kb=$F need_kb=$((K*2+5242880))"; exit 1; }; rsync -a --partial -e "ssh $O" $S:$B/$L/ $D/biap-global-$L.partial/ && rm -rf $D/biap-global-$L && mv $D/biap-global-$L.partial $D/biap-global-$L && ls -1d $D/biap-global-????-??-?? | sort | head -n -2 | xargs -r rm -rf -- && echo "$(date -u +%FT%TZ) OK biap-global-$L"' >> /var/log/biap-liara-pull.log 2>&1
